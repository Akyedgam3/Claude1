'use strict';
const $ = (s, r = document) => r.querySelector(s);
const esc = (v) => String(v ?? '').replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
const app = $('#app');
let me = null, S = null, tab = 'home', q = '';
const LINK = location.origin;

async function api(path, body) {
  const r = await fetch(path, body === undefined ? { credentials: 'same-origin' } : { method: 'POST', credentials: 'same-origin', headers: { 'Content-Type': 'application/json', 'X-Wazn': '1' }, body: JSON.stringify(body) });
  const j = await r.json().catch(() => ({}));
  if (j.needOtp) { const e = new Error('اكتب رمز التحقق من تطبيق المصادقة'); e.needOtp = true; throw e; }
  if (r.status === 401 && me) { me = null; render(); }
  if (!r.ok) throw new Error(j.error || 'خطأ');
  return j;
}
function toast(m, bad) {
  const t = document.createElement('div');
  t.className = 'toast' + (bad ? ' bad' : '');
  t.textContent = m;
  document.body.appendChild(t);
  setTimeout(() => t.remove(), bad ? 6000 : 3500);
}
async function run(fn, ok) {
  try { const r = await fn(); if (ok) toast(ok); await load(); return r; } catch (e) { toast(e.message, true); }
}
const copy = (t, m = 'تم النسخ') => navigator.clipboard?.writeText(t).then(() => toast(m), () => toast('ما قدرت انسخ', true));
const fmtDate = (d) => (d ? String(d).slice(0, 10) : 'دائم');
const fmtTime = (d) => String(d ?? '').slice(0, 16).replace('T', ' ');
const money = (n, cur = 'USD') => (cur === 'SYP' ? Number(n).toLocaleString('en-US') + ' ل.س' : '$' + Number(n).toFixed(2));
const kindName = (k) => (k === 'lifetime' ? 'دائم' : 'سنوي');
const BUCKETS = [['annual', 'سنوي'], ['lifetime', 'دائم'], ['annualMobile', 'سنوي + جوال'], ['lifetimeMobile', 'دائم + جوال']];
const tierLabel = (t) => `${t.name} ${kindName(t.kind)}${t.mobile ? ' + جوال' : ''} — ${t.count} رخصة (${money(t.price, 'SYP')})`;
const daysLeft = (c) => (c.expiresAt ? Math.ceil((new Date(c.expiresAt) - Date.now()) / 86400000) : null);
const statusTag = (c) => `<span class="tag ${esc(c.status)}">${esc({ active: 'فعّال', suspended: 'موقوف', revoked: 'ملغى' }[c.status] ?? c.status)}</span>`;

function modal(html, onMount) {
  const bg = document.createElement('div');
  bg.className = 'modal-bg';
  bg.innerHTML = `<div class="modal">${html}</div>`;
  bg.addEventListener('click', (e) => { if (e.target === bg) bg.remove(); });
  document.body.appendChild(bg);
  bg.close = () => bg.remove();
  onMount?.(bg);
  return bg;
}

'<span class="tag mob">+ جوال</span>' : ''}<div class="muted">${esc(k.name)} · ${esc(S.plans?.[k.plan]?.nameAr ?? (k.plan === 'lifetime' ? function bindCards(el, voidPath) {
  el.querySelectorAll('[data-k]').forEach((b) => (b.onclick = async () => {
    const k = S.cards.find((x) => x.id === b.dataset.id);
    if (b.dataset.k === 'show') return showCard(k);
    if (confirm(`إلغاء الكود ${k.code}؟${voidPath.includes('agent') ? ' الرخصة بترجع لرصيدك.' : ''}`)) await run(() => api(voidPath.replace(':id', k.id), {}), 'انلغى الكود');
  }));
}

  app.innerHTML = `<div class="login card"><h1>وزن</h1><p class="muted" style="text-align:center">${needsSetup ? 'إعداد أول مرة: حساب المدير' : 'لوحة الإدارة والوكلاء'}</p>
    <form id="f" class="list">
      ${needsSetup ? '<label>رمز الإعداد<input name="code" required autocomplete="off" /></label>' : ''}
      <label>اسم المستخدم<input name="username" required autocomplete="username" dir="ltr" autocapitalize="none" /></label>
      <label>كلمة السر<input name="password" type="password" required autocomplete="${needsSetup ? 'new-password' : 'current-password'}" /></label>
      <label id="otpl" class="hidden">رمز التحقق (6 أرقام من تطبيق المصادقة)<input name="otp" inputmode="numeric" autocomplete="one-time-code" dir="ltr" maxlength="6" /></label>
      ${needsSetup ? '<p class="muted">كلمة سر قوية، 10 حروف عالأقل. هالحساب بيتحكم بكل التراخيص.</p>' : ''}
      <button class="primary">${needsSetup ? 'إنشاء الحساب' : 'دخول'}</button>
    </form></div>`;
  $('#f').onsubmit = async (e) => {
    e.preventDefault();
    const body = Object.fromEntries(new FormData(e.target));
    try { await api(needsSetup ? '/api/setup' : '/api/login', body); await boot(); } catch (err) {
      if (err.needOtp) { $('#otpl').classList.remove('hidden'); $('#otpl input').focus(); }
      toast(err.message, !err.needOtp);
    }
  };
}

async function boot() {
  const r = await api('/api/me');
  me = r.user;
  if (!me) return renderLogin(r.needsSetup);
  tab = me.role === 'owner' ? 'home' : 'agent';
  await load();
}
async function load() {
  if (!me) return;
  S = await api(me.role === 'owner' ? '/api/state' : '/api/agent/state');
  render();
}
function render() {
  if (!me) return boot();
  const owner = me.role === 'owner';
  app.innerHTML = `<header><b>وزن</b><span>${owner ? 'لوحة المدير' : 'لوحة الوكيل'}</span><span class="who">${esc(me.name)}</span><button class="small" id="pw">كلمة السر</button><button class="small" id="out">خروج</button></header><main id="m"></main>`;
  $('#out').onclick = () => run(() => api('/api/logout', {})).then(() => { me = null; renderLogin(false); });
  $('#pw').onclick = changePassword;
  const m = $('#m');
  if (!owner) return renderAgent(m);
  const tabs = [['home', 'الرئيسية'], ['agents', 'الوكلاء'], ['customers', 'الزباين'], ['updates', 'التحديثات'], ['settings', 'الإعدادات']];
  m.innerHTML = `<nav class="tabs">${tabs.map(([k, n]) => `<button class="${tab === k ? 'on' : ''}" data-t="${k}">${n}</button>`).join('')}</nav><div id="body"></div>`;
  m.querySelectorAll('[data-t]').forEach((b) => (b.onclick = () => { tab = b.dataset.t; q = ''; render(); }));
  ({ home: ownerHome, agents: ownerAgents, customers: ownerCustomers, updates: ownerUpdates, settings: ownerSettings })[tab]($('#body'));
}

}

function publishBanner() {
  const p = S.publish;
  if (!S.settings.hasKey) return `<div class="banner bad">ما في مفتاح توقيع لسا: روح على <b>الإعدادات</b> وارفع <span class="mono">license-private.pem</span>.</div>`;
  if (!S.settings.hasGithub) return `<div class="banner">النشر التلقائي مش شغّال: ضيف رمز GitHub من <b>الإعدادات</b>.</div>`;
  if (S.pending) return `<div class="banner">في تغييرات عم تنتشر خلال ثواني…</div>`;
  if (p && !p.ok) return `<div class="banner bad">${esc(p.message)}</div>`;
  return p ? `<div class="banner ok">آخر نشر تلقائي: ${esc(fmtTime(p.at))} ✓</div>` : '';
}
function ownerHome(el) {
  const now = Date.now();
  const twoFa = me.twoFactor ? '' : `<div class="banner bad">🔒 حسابك بكلمة سر بس. فعّل <b>التحقق بخطوتين</b> من الإعدادات: حتى لو حدا عرف كلمة السر، ما بيقدر يدخل بدون جوالك.</div>`;
  const active = S.customers.filter((c) => c.status === 'active');
  const soon = active.filter((c) => { const d = daysLeft(c); return d !== null && d >= 0 && d <= 30; });
  el.innerHTML = twoFa + publishBanner() + `<div class="stats">
    <div class="stat"><span>الزباين الفعّالين</span><b>${active.length}</b></div>
    <div class="stat"><span>بينتهوا خلال شهر</span><b>${soon.length}</b></div>
    <div class="stat"><span>الوكلاء</span><b>${S.agents.filter((a) => a.active).length}</b></div>
    <div class="stat"><span>رصيد عند الوكلاء</span><b>${S.agents.reduce((s, a) => s + a.balance.left, 0)}</b></div>
    <div class="stat"><span>تفعيلات آخر 7 أيام</span><b>${S.customers.filter((c) => now - new Date(c.issuedAt) < 7 * 86400000).length}</b></div>
  </div>
  <div class="card"><h2>آخر النشاط</h2><ul class="log">${S.log.slice(0, 80).map((l) => `<li><b>${esc(l.by)}</b>: ${esc(l.action)} <span class="muted">· ${esc(fmtTime(l.at))}</span></li>`).join('') || '<li class="muted">لسا ما في نشاط</li>'}</ul></div>`;

async function agentAction(act, id) {
  const a = S.agents.find((x) => x.id === id);
  if (act === 'pkg') {
    const t = S.tiers.find((x) => x.id === $(`#tier-${id}`).value);
    if (confirm(`إضافة «${tierLabel(t)}» لرصيد ${a.name}؟`)) await run(() => api(`/api/agents/${id}/package`, { tier: t.id }), 'انضافت الخطة');
  } else if (act === 'custom') {
    const m = modal(`<h2>باقة مخصصة لـ ${esc(a.name)}</h2><form class="list" id="f"><label>النوع<select name="b">${BUCKETS.map(([k, n]) => `<option value="${k}">${n}</option>`).join('')}</select></label>
      <label>عدد الرخص<input name="count" type="number" min="1" value="10" required /></label><label>المبلغ المدفوع (ل.س جديدة)<input name="price" type="number" min="0" value="0" /></label><label>ملاحظة<input name="note" /></label>
      <div class="row"><button class="primary">إضافة</button><button type="button" id="x">إلغاء</button></div></form>`);
    $('#x', m).onclick = m.close;
    $('#f', m).onsubmit = async (e) => {
      e.preventDefault();
      const f = Object.fromEntries(new FormData(e.target));
      const r = await run(() => api(`/api/agents/${id}/package`, { count: f.count, price: f.price, note: f.note, kind: f.b.startsWith('lifetime') ? 'lifetime' : 'annual', mobile: f.b.endsWith('Mobile') }), 'انضافت الباقة');
      if (r) m.close();
    };
  } else if (act === 'reset') {
    if (!confirm(`كلمة سر جديدة لـ ${a.name}؟ القديمة بتبطل فوراً.`)) return;
    const r = await run(() => api(`/api/agents/${id}/reset`, {}));
    if (r) sendCreds(a, r.username, r.password, false);
  } else if (act === 'login') {
    const username = prompt('اسم مستخدم للوكيل (إنكليزي صغير):', '');
    if (!username) return;
    const r = await run(() => api(`/api/agents/${id}/login`, { username }));
    if (r) sendCreds(a, r.username, r.password, true);
  } else if (act === 'edit') {
    const m = modal(`<h2>تعديل ${esc(a.name)}</h2><form class="list" id="f"><label>الاسم<input name="name" value="${esc(a.name)}" /></label><label>الموبايل<input name="phone" dir="ltr" value="${esc(a.phone)}" /></label><label>المدينة<input name="city" value="${esc(a.city)}" /></label><label>ملاحظات<input name="notes" value="${esc(a.notes)}" /></label><div class="row"><button class="primary">حفظ</button><button type="button" id="x">إلغاء</button></div></form>`);
    $('#x', m).onclick = m.close;
    $('#f', m).onsubmit = async (e) => { e.preventDefault(); const r = await run(() => api(`/api/agents/${id}/update`, Object.fromEntries(new FormData(e.target))), 'تم الحفظ'); if (r) m.close(); };
  } else if (act === 'stop') {

async function customerAction(act, id, owner) {
  const c = S.customers.find((x) => x.id === id);
  if (act === 'key') return showKey(c);
  if (act === 'phone') {
    const machineId = prompt('بصمة جوال الزبون (من شاشة التفعيل بالتطبيق):', '');
    if (!machineId) return;
    const r = await run(() => api(owner ? `/api/customers/${id}/phone` : `/api/agent/customers/${id}/phone`, { machineId }), 'انعمل مفتاح الجوال');
    if (r) showKey(r);
  } else if (act === 'renew') {
    if (!confirm(`تجديد ${c.name} سنة؟ بتنخصم رخصة من رصيدك.`)) return;
    const r = await run(() => api(`/api/agent/customers/${id}/renew`, {}), 'تم التجديد');
    if (r) showKey(r, '<p class="muted">إذا البرنامج عند الزبون متصل بالنت ما بيحتاج المفتاح الجديد، بيتجدد لحالو.</p>');
  } else if (act === 'extend') {
    const m = modal(`<h2>تمديد ${esc(c.name)}</h2><form class="list" id="f"><label>المدة<select name="plan">${Object.entries(S.plans).map(([k, p]) => `<option value="${k}" ${k === 'annual' ? 'selected' : ''}>${esc(p.nameAr)}</option>`).join('')}</select></label><div class="row"><button class="primary">تمديد</button><button type="button" id="x">إلغاء</button></div></form>`);
    $('#x', m).onclick = m.close;
    $('#f', m).onsubmit = async (e) => { e.preventDefault(); const r = await run(() => api(`/api/customers/${id}/extend`, Object.fromEntries(new FormData(e.target))), 'تم التمديد، وبينتشر لحالو'); if (r) m.close(); };
  } else if (act === 'suspend' || act === 'revoke') {
    const message = prompt(act === 'revoke' ? 'سبب الإلغاء (بيطلع للزبون):' : 'رسالة بتطلع للزبون:', act === 'suspend' ? 'تم إيقاف البرنامج لعدم تسديد الاشتراك. يرجى التواصل مع البائع.' : '');

  } else if (act === 'activate') {
    await run(() => api(`/api/customers/${id}/status`, { status: 'active' }), 'رجع فعّال');
  } else if (act === 'cloud') {
    await run(() => api(`/api/customers/${id}/cloud`, { on: !c.cloud }), c.cloud ? 'انطفى الاتصال عن بعد' : 'انفتح الاتصال عن بعد');

  }
}

    const b2 = $('#cp2'); if (b2 && r?.phone) b2.onclick = () => copy(r.phone.code, 'نسخت كود الجوال');
  };
  $('#ak').onsubmit = async (e) => {
    e.preventDefault();
    const f = Object.fromEntries(new FormData(e.target));
    const k = await run(() => api('/api/agent/cards', { name: f.name, phone: f.phone, plan: f.bucket.startsWith('lifetime') ? 'lifetime' : 'annual', mobile: f.bucket.endsWith('Mobile') }));

  };

}
