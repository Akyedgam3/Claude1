async function api(path, body) {
  const r = await fetch(path, body === undefined ? { credentials: 'same-origin' } : { method: 'POST', credentials: 'same-origin', headers: { 'Content-Type': 'application/json', 'X-Wazn': '1' }, body: JSON.stringify(body) });
  const j = await r.json().catch(() => ({}));
  if (j.needOtp) { const e = new Error('اكتب رمز التحقق من تطبيق المصادقة'); e.needOtp = true; throw e; }
  if (r.status === 401 && me) { me = null; render(); }
  if (!r.ok) throw new Error(j.error || 'خطأ');
  return j;
}
function toast(m, bad) {
  const t = document.createElement('div');
  t.className = 'toast' + (bad ? ' bad' : '');
  t.textContent = m;
  document.body.appendChild(t);
  setTimeout(() => t.remove(), bad ? 6000 : 3500);
}
async function run(fn, ok) {
  try { const r = await fn(); if (ok) toast(ok); await load(); return r; } catch (e) { toast(e.message, true); }
}

function modal(html, onMount) {
  const bg = document.createElement('div');
  bg.className = 'modal-bg';
  bg.innerHTML = `<div class="modal">${html}</div>`;
  bg.addEventListener('click', (e) => { if (e.target === bg) bg.remove(); });
  document.body.appendChild(bg);
  bg.close = () => bg.remove();
  onMount?.(bg);
  return bg;
}

async function boot() {
  const r = await api('/api/me');
  me = r.user;
  if (!me) return renderLogin(r.needsSetup);
  tab = me.role === 'owner' ? 'home' : 'agent';
  await load();
}
async function load() {
  if (!me) return;
  S = await api(me.role === 'owner' ? '/api/state' : '/api/agent/state');
  render();
}
function render() {

  const owner = me.role === 'owner';

/** صورة الفاتورة (PNG) مرسومة على canvas بخط Cairo. */
async function invoiceImage(v) {
  await Promise.all(['400', '700', '900'].map((w) => document.fonts.load(`${w} 40px Cairo`).catch(() => null)));
  const rows = [
    ['نوع الاشتراك', `${v.planName}${v.mobile ? ' (كمبيوتر + جوال)' : ''}`],
    ['تاريخ الاشتراك', dmy(v.date)],
    ['صالح لغاية', v.until ? dmy(v.until) : 'دائم بدون انتهاء'],
    ['عدد الأجهزة', String(v.devices)],
  ];
  if (v.code) rows.push(['كود التفعيل', v.code]);
  const W = 1080, P = 70, H = 515 + rows.length * 92 + 15 + 45 + 150 + (v.note ? 50 : 0) + 290 + 130;
  const cv = document.createElement('canvas');
  cv.width = W; cv.height = H;
  const x = cv.getContext('2d');
  const font = (w, s) => (x.font = `${w} ${s}px Cairo, sans-serif`);
  const text = (t, px, py, o = {}) => {
    x.save(); font(o.w ?? 700, o.s ?? 34); x.fillStyle = o.c ?? '#1d2433'; x.textAlign = o.a ?? 'right'; x.direction = o.ltr ? 'ltr' : 'rtl'; x.fillText(String(t), px, py, o.max); x.restore();
  };
  const rr = (px, py, w, h, r) => { x.beginPath(); x.roundRect(px, py, w, h, r); };
  const R = W - P;
  // الخلفية والرأس
  x.fillStyle = '#f4f6f8'; x.fillRect(0, 0, W, H);
  const g = x.createLinearGradient(0, 0, W, 360); g.addColorStop(0, '#0b5546'); g.addColorStop(1, '#14957c');
  x.fillStyle = g; x.fillRect(0, 0, W, 360);
  x.globalAlpha = 0.08; x.fillStyle = '#fff';
  for (let i = 0; i < 6; i++) { x.beginPath(); x.arc(120 + i * 40, 60 + i * 30, 160 + i * 50, 0, Math.PI * 2); x.fill(); }
  x.globalAlpha = 1;
  rr(R - 120, 70, 120, 120, 30); x.fillStyle = '#fff'; x.fill();
  text('وزن', R - 60, 150, { s: 54, w: 900, c: '#0f6e5c', a: 'center' });
  text('برنامج وزن', R - 150, 122, { s: 40, w: 900, c: '#fff' });
  text('للمحاسبة وإدارة المحلات', R - 150, 170, { s: 26, w: 700, c: 'rgba(255,255,255,.85)' });
  text('فاتورة اشتراك', P, 125, { s: 50, w: 900, c: '#fff', a: 'left' });
  text(`رقم ${v.n}  ·  ${dmy(v.date)}`, P, 175, { s: 28, w: 700, c: 'rgba(255,255,255,.85)', a: 'left' });
  // بطاقة العميل
  rr(P, 250, W - 2 * P, 190, 28); x.fillStyle = '#fff'; x.shadowColor = 'rgba(15,40,30,.12)'; x.shadowBlur = 30; x.shadowOffsetY = 8; x.fill(); x.shadowColor = 'transparent';
  text('العميل', R - 40, 310, { s: 26, c: '#6b7385' });
  text(v.name, R - 40, 370, { s: 46, w: 900, max: v.phone ? 520 : 860 });
  if (v.phone) text(v.phone, P + 40, 370, { s: 34, w: 700, c: '#0f6e5c', a: 'left', ltr: true });
  if (v.phone) text('الرقم', P + 40, 310, { s: 26, c: '#6b7385', a: 'left' });
  // البنود
  let y = 500;
  rr(P, y, W - 2 * P, rows.length * 92 + 30, 28); x.fillStyle = '#fff'; x.fill();
  y += 15;
  rows.forEach(([k, val], i) => {
    if (i) { x.fillStyle = '#e3e6eb'; x.fillRect(P + 40, y, W - 2 * P - 80, 2); }
    text(k, R - 40, y + 60, { s: 30, w: 700, c: '#6b7385' });
    const isCode = k === 'كود التفعيل';
    text(val, P + 40, y + 60, { s: 34, w: 900, max: 600, a: 'left', ltr: isCode, c: isCode ? '#0f6e5c' : '#1d2433' });

function bindCustomers(el, owner) {

async function customerAction(act, id, owner) {
  const c = S.customers.find((x) => x.id === id);
  if (act === 'key') return showKey(c);
  if (act === 'phone') {
    const machineId = prompt('بصمة جوال الزبون (من شاشة التفعيل بالتطبيق):', '');
    if (!machineId) return;
    const r = await run(() => api(owner ? `/api/customers/${id}/phone` : `/api/agent/customers/${id}/phone`, { machineId }), 'انعمل مفتاح الجوال');
    if (r) showKey(r);
  } else if (act === 'renew') {
    if (!confirm(`تجديد ${c.name} سنة؟ بتنخصم رخصة من رصيدك.`)) return;
    const r = await run(() => api(`/api/agent/customers/${id}/renew`, {}), 'تم التجديد');
    if (r) showKey(r, '<p class="muted">إذا البرنامج عند الزبون متصل بالنت ما بيحتاج المفتاح الجديد، بيتجدد لحالو.</p>');
  } else if (act === 'lifetime') {
    if (!confirm(`تحويل ${c.name} لاشتراك دائم؟ بتنخصم رخصة من رصيدك الدائم${c.mobile ? ' مع الجوال' : ''}.`)) return;
    await run(() => api(`/api/agent/customers/${id}/lifetime`, {}), 'صار اشتراكه دائم');
  } else if (act === 'extend') {
    const ag = c.agentId ? S.agents?.find((a) => a.id === c.agentId) : null;
    const lifetime = c.plan === 'lifetime';
    const m = modal(`<h2>تمديد أو تحويل ${esc(c.name)}</h2><form class="list" id="f">${lifetime ? '<p class="muted">اشتراكه دائم أصلاً.</p>' : ''}<label>المدة<select name="plan">${Object.entries(S.plans).map(([k, p]) => `<option value="${k}" ${k === 'annual' ? 'selected' : ''}>${esc(p.nameAr)}${k === 'lifetime' ? ' (تحويل لدائم)' : ''}</option>`).join('')}</select></label>
      ${ag ? `<label class="row" style="gap:6px;color:var(--ink)"><input type="checkbox" name="charge" checked style="width:auto" /> خصم رخصة من رصيد الوكيل ${esc(ag.name)} (سنة من السنوي، دائم من الدائم)</label>` : ''}
      <div class="row"><button class="primary">تمديد</button><button type="button" id="x">إلغاء</button></div></form>`);
    $('#x', m).onclick = m.close;
    $('#f', m).onsubmit = async (e) => { e.preventDefault(); const f = Object.fromEntries(new FormData(e.target)); const r = await run(() => api(`/api/customers/${id}/extend`, { plan: f.plan, ...(ag && !f.charge ? { free: true } : {}) }), f.plan === 'lifetime' ? 'صار اشتراكه دائم، وبينتشر لحالو' : 'تم التمديد، وبينتشر لحالو'); if (r) m.close(); };
  } else if (act === 'suspend' || act === 'revoke') {
    const message = prompt(act === 'revoke' ? 'سبب الإلغاء (بيطلع للزبون):' : 'رسالة بتطلع للزبون:', act === 'suspend' ? 'تم إيقاف البرنامج لعدم تسديد الاشتراك. يرجى التواصل مع البائع.' : '');
    if (message === null) return;
    if (act === 'revoke' && !confirm('إلغاء نهائي؟')) return;
    await run(() => api(`/api/customers/${id}/status`, { status: act === 'revoke' ? 'revoked' : 'suspended', message }), 'تم، وبينتشر لحالو');
  } else if (act === 'activate') {
    await run(() => api(`/api/customers/${id}/status`, { status: 'active' }), 'رجع فعّال');
  } else if (act === 'cloud') {
    await run(() => api(`/api/customers/${id}/cloud`, { on: !c.cloud }), c.cloud ? 'انطفى الاتصال عن بعد' : 'انفتح الاتصال عن بعد');
  } else if (act === 'edit') {
    const m = modal(`<h2>تعديل ${esc(c.name)}</h2><form class="list" id="f"><label>الاسم<input name="name" value="${esc(c.name)}" /></label><label>الموبايل<input name="phone" dir="ltr" value="${esc(c.phone)}" /></label><label>الأجهزة<input name="devices" type="number" min="1" max="50" value="${c.devices}" /></label><label>أيام العمل بدون نت<input name="offlineDays" type="number" min="1" max="60" value="${c.offlineDays}" /></label><label>بصمة جديدة (إذا غيّر الجهاز)<input name="machineId" class="mono" placeholder="${esc(c.machineId)}" /></label><label>ملاحظات<input name="notes" value="${esc(c.notes)}" /></label><div class="row"><button class="primary">حفظ وإصدار مفتاح جديد</button><button type="button" id="x">إلغاء</button></div></form>`);
    $('#x', m).onclick = m.close;
    $('#f', m).onsubmit = async (e) => { e.preventDefault(); const r = await run(() => api(`/api/customers/${id}/update`, Object.fromEntries(new FormData(e.target)))); if (r) { m.close(); showKey(r); } };
  }
}
